import {Dimensions, PixelRatio} from 'react-native';
import {moderateScale, scale, verticalScale} from 'react-native-size-matters';
import {
  widthPercentageToDP as wp,
  heightPercentageToDP as hp,
} from 'react-native-responsive-screen';

// Moderate scale — good for font sizes, padding, margin
export const normalizeModerately = (size: number, factor = 0.5): number => {
  return PixelRatio.roundToNearestPixel(moderateScale(size, factor));
};

// Horizontal / width scaling
export const normalizeWidth = (size: number): number => {
  return PixelRatio.roundToNearestPixel(scale(size));
};

// Vertical / height scaling
export const normalizeHeight = (size: number): number => {
  return PixelRatio.roundToNearestPixel(verticalScale(size));
};

// Percentage-based width (pass string like '50%')
export const widthPercentage = (size: string): number => {
  return wp(size);
};

// Percentage-based height
export const heightPercentage = (size: string): number => {
  return hp(size);
};

export const screenWidth: number  = Dimensions.get('window').width;
export const screenHeight: number = Dimensions.get('window').height;

import {
  createNavigationContainerRef,
  CommonActions,
  StackActions,
} from '@react-navigation/native';

export const navigationRef = createNavigationContainerRef();

// Navigate to a screen (forward)
export async function navigate(routeName: string, params?: object) {
  await navigationRef.isReady();
  if (navigationRef.isReady()) {
    navigationRef.dispatch(CommonActions.navigate(routeName, params));
  }
}

// Reset stack and go to a screen (e.g. after login)
export async function resetAndNavigate(routeName: string) {
  await navigationRef.isReady();
  if (navigationRef.isReady()) {
    navigationRef.dispatch(
      CommonActions.reset({
        index: 0,
        routes: [{name: routeName}],
      }),
    );
  }
}

// Go back
export async function goBack() {
  await navigationRef.isReady();
  if (navigationRef.isReady()) {
    navigationRef.dispatch(CommonActions.goBack());
  }
}

// Push a new screen on top of the stack (allows duplicates)
export async function push(routeName: string, params?: object) {
  await navigationRef.isReady();
  if (navigationRef.isReady()) {
    navigationRef.dispatch(StackActions.push(routeName, params));
  }
}

// Await navigation readiness before app action
export async function prepareNavigation() {
  await navigationRef.isReady();
}
